def compare(actual,expected,columns):
    assert len(actual)==len(expected)
    worst={}
    for c in columns:
        a=pd.to_numeric(actual[c],errors='coerce').to_numpy(dtype=float)
        b=pd.to_numeric(expected[c],errors='coerce').to_numpy(dtype=float)
        np.testing.assert_array_equal(np.isnan(a),np.isnan(b),err_msg=c)
        if c.endswith(('_code','_flag')) or c.startswith('supertrend_direction'):
            np.testing.assert_array_equal(a,b,err_msg=c)
        else:
            np.testing.assert_allclose(a,b,rtol=1e-8,atol=1e-8,equal_nan=True,err_msg=c)
        valid=np.isfinite(a)&np.isfinite(b)
        worst[c]=float(np.max(np.abs(a[valid]-b[valid]))) if valid.any() else 0
    return sorted(worst.items(),key=lambda kv:kv[1],reverse=True)[:5]

for instrument,timeframe in [('FUT_BR','15m'),('FUT_BR','1h'),('FUT_GOLD','4h'),('FUT_BR','1d')]:
    print('START',instrument,timeframe,flush=True)
    table=DeltaTable(str(source)).to_pyarrow_table(filters=[('instrument_id','=',instrument),('timeframe','=',timeframe)])
    if len(table)<700:
        print('SKIP insufficient warm history',len(table),flush=True)
        continue
    table=table.sort_by([('ts','ascending')])
    # Retired source snapshots contain a late legacy epoch=0 segment after
    # historical epochs. Tail admission rejects such nonmonotonic inputs.
    epochs=np.asarray(table['roll_epoch'].to_pylist(),dtype=int)
    drops=np.flatnonzero(np.diff(epochs)<0)+1

    left,right=max(zip(edges[:-1],edges[1:]),key=lambda pair:pair[1]-pair[0])
    excluded_nonmonotonic=len(table)-(right-left)
    table=table.slice(left,right-left)

    bars=list(bars[:])
    # Pick an append within the final epoch; rolls themselves are tested fallback.
    epoch=bars[-1].roll_epoch

    if appended<1:continue

    options=dict(dataset_version=bars[0].dataset_version,indicator_set_version='tail-proof',

    start=time.perf_counter()
